  const RESULTS_KEY = "flow_scene_results_v1";
  const MAX_SCENE_ATTEMPTS = 3;            // Automatically skips scene after 3 failed attempts
  const GENERATE_START_DELAY_MS = 8000;
  const POLL_INTERVAL_MS = 2000;
  const POLL_MAX_TRIES = 35;
  const STABLE_RECHECKS = 2;
  const STABLE_RECHECK_DELAY_MS = 1800;
  const RETRY_DELAY_MS = 6000;
  const MIN_TILE_WIDTH = 220;
  const MIN_TILE_HEIGHT = 130;

  const delay = (ms) => new Promise((res) => setTimeout(res, ms));

  function loadResults() {
    try {
      return JSON.parse(localStorage.getItem(RESULTS_KEY) || "{}");
    } catch {
      return {};
    }
  }

  function saveResults(results) {
    localStorage.setItem(RESULTS_KEY, JSON.stringify(results));
  }

  function downloadResultsSummary(results) {
    const blob = new Blob([JSON.stringify(results, null, 2)], { type: "application/json" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = "flow_results.json";
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(a.href);
  }

  function findInputElement() {
    return (
      document.querySelector('div[contenteditable="true"]') ||
      document.querySelector("textarea") ||
      document.querySelector('[role="textbox"]')
    );
  }

  function injectTextReal(el, text) {
    el.focus();
    el.click();

    const proto = el.isContentEditable
      ? window.HTMLElement.prototype
      : window.HTMLTextAreaElement.prototype || window.HTMLInputElement.prototype;
    const desc = Object.getOwnPropertyDescriptor(proto, "value");
    if (desc && desc.set) {
      desc.set.call(el, text);
    } else {
      el.innerText = text;
    }

    el.dispatchEvent(new Event("focus", { bubbles: true }));
    el.dispatchEvent(new InputEvent("beforeinput", { bubbles: true, inputType: "insertText", data: text }));
    el.dispatchEvent(new InputEvent("input", { bubbles: true, data: text }));
    el.dispatchEvent(new Event("change", { bubbles: true }));
    el.dispatchEvent(new KeyboardEvent("keydown", { key: " ", code: "Space", bubbles: true }));
    el.dispatchEvent(new KeyboardEvent("keyup", { key: " ", code: "Space", bubbles: true }));
  }

  function triggerSubmit(el) {
    el.dispatchEvent(new KeyboardEvent("keydown", { key: "Enter", code: "Enter", keyCode: 13, which: 13, bubbles: true }));
    el.dispatchEvent(new KeyboardEvent("keypress", { key: "Enter", code: "Enter", keyCode: 13, which: 13, bubbles: true }));
    el.dispatchEvent(new KeyboardEvent("keyup", { key: "Enter", code: "Enter", keyCode: 13, which: 13, bubbles: true }));
  }

  function getAllDeepElements(root = document.body) {
    let elements = [];
    const walk = (node) => {
      if (!node) return;
      elements.push(node);
      if (node.shadowRoot) walk(node.shadowRoot);
      for (const child of node.children || []) walk(child);
    };
    walk(root);
    return elements;
  }

  function isLikelyPlaceholder(el, src) {
    if (src && src.startsWith("data:")) return true;

    const style = window.getComputedStyle(el);
    if (style.filter && style.filter !== "none" && /blur/i.test(style.filter)) return true;

    let parent = el.parentElement;
    let depth = 0;
    while (parent && depth < 4) {
      const pStyle = window.getComputedStyle(parent);
      if (pStyle.filter && /blur/i.test(pStyle.filter)) return true;
      const cls = (parent.className || "").toString();
      if (/placeholder|skeleton|shimmer|blurhash/i.test(cls)) return true;
      parent = parent.parentElement;
      depth++;
    }

    const cls = (el.className || "").toString();
    if (/placeholder|skeleton|shimmer|blurhash/i.test(cls)) return true;

    return false;
  }

  function getTopLeftTile() {
    const allEls = getAllDeepElements();
    const mediaCandidates = [];

    for (const el of allEls) {
      if (!el.getBoundingClientRect) continue;
      const rect = el.getBoundingClientRect();
      const inSidebar = el.closest("aside") || el.closest('[class*="dock"]') || el.closest('[class*="chat"]');
      if (inSidebar) continue;
      if (rect.width < MIN_TILE_WIDTH || rect.height < MIN_TILE_HEIGHT) continue;

      if (el.tagName === "IMG" && el.src) {
        if (isLikelyPlaceholder(el, el.src)) continue;
        if (!el.complete || el.naturalWidth < MIN_TILE_WIDTH) continue;
        mediaCandidates.push({ el, src: el.src, rect, type: "img" });
      } else if (el.tagName === "CANVAS") {
        if (el.width < MIN_TILE_WIDTH || el.height < MIN_TILE_HEIGHT) continue;
        mediaCandidates.push({ el, rect, type: "canvas" });
      } else {
        const bg = window.getComputedStyle(el).backgroundImage;
        if (bg && bg.includes("url(")) {
          const m = bg.match(/url\(["']?(.*?)["']?\)/);
          if (m && m[1] && !isLikelyPlaceholder(el, m[1])) {
            mediaCandidates.push({ el, src: m[1], rect, type: "bg" });
          }
        }
      }
    }

    if (mediaCandidates.length === 0) return null;

    mediaCandidates.sort((a, b) => {
      if (Math.abs(a.rect.top - b.rect.top) > 50) return a.rect.top - b.rect.top;
      return a.rect.left - b.rect.left;
    });

    return mediaCandidates[0];
  }

  async function waitForStableNewTile(initialKey) {
    let candidate = null;
    let stableHits = 0;

    for (let i = 1; i <= POLL_MAX_TRIES; i++) {
      await delay(POLL_INTERVAL_MS);
      const tile = getTopLeftTile();
      const key = tile ? tile.src || "canvas_element" : null;

      if (tile && key !== initialKey) {
        if (candidate && candidate.key === key) {
          stableHits++;
        } else {
          candidate = { tile, key };
          stableHits = 1;
        }

        if (stableHits >= STABLE_RECHECKS) {
          await delay(STABLE_RECHECK_DELAY_MS);
          const confirmTile = getTopLeftTile();
          const confirmKey = confirmTile ? confirmTile.src || "canvas_element" : null;
          if (confirmKey === key) {
            return confirmTile;
          }
          candidate = null;
          stableHits = 0;
        }
      } else {
        console.log(`  ...waiting for tile render (${i}/${POLL_MAX_TRIES})`);
      }
    }
    return null;
  }

  function triggerBlobDownload(blob, filename) {
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(a.href);
  }

  async function tryFetchDownload(url, filename, credentials) {
    const res = await fetch(url, { credentials });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const blob = await res.blob();
    if (blob.size < 3000) throw new Error(`Blob size ${blob.size} bytes — error or placeholder.`);
    if (blob.type && !blob.type.startsWith("image/") && blob.type !== "application/octet-stream") {
      throw new Error(`Blob type "${blob.type}" is not an image.`);
    }
    triggerBlobDownload(blob, filename);
    return true;
  }

  async function tryCanvasReRender(url, filename) {
    return new Promise((resolve) => {
      const img = new Image();
      img.crossOrigin = "anonymous";
      img.onload = () => {
        try {
          const canvas = document.createElement("canvas");
          canvas.width = img.naturalWidth || 1280;
          canvas.height = img.naturalHeight || 720;
          const ctx = canvas.getContext("2d");
          ctx.drawImage(img, 0, 0);
          canvas.toBlob((b) => {
            if (!b) return resolve(false);
            triggerBlobDownload(b, filename);
            resolve(true);
          }, "image/png");
        } catch (err) {
          resolve(false);
        }
      };
      img.onerror = () => resolve(false);
      img.src = url;
    });
  }

  async function downloadMasterImage(tile, filename) {
    if (tile.type === "canvas") {
      try {
        const dataUrl = tile.el.toDataURL("image/png");
        const a = document.createElement("a");
        a.href = dataUrl;
        a.download = filename;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        return true;
      } catch (err) {
        console.warn(`Canvas export failed: ${err.message}`);
      }
    }

    const url = tile.src;
    if (!url) return false;

    if (url.startsWith("blob:")) {
      try {
        return await tryFetchDownload(url, filename, "same-origin");
      } catch (err) {}
    }

    const attempts = [
      () => tryFetchDownload(url, filename, "include"),
      () => tryFetchDownload(url, filename, "omit"),
    ];

    for (const attemptFn of attempts) {
      try {
        const ok = await attemptFn();
        if (ok) return true;
      } catch (err) {}
    }

    const canvasOk = await tryCanvasReRender(url, filename);
    if (canvasOk) return true;

    return false;
  }

  // ---------------------------------------------------------------------
  // SCENE EXECUTION (WITH AUTO-SKIP)
  // ---------------------------------------------------------------------
  async function processScene(sceneId, promptText) {
    let attempt = 0;

    while (attempt < MAX_SCENE_ATTEMPTS) {
      attempt++;
      console.log(`%c[${sceneId}] Attempt ${attempt}/${MAX_SCENE_ATTEMPTS}`, "color:#00e5ff;font-weight:bold;");

      const initialTile = getTopLeftTile();
      const initialKey = initialTile ? initialTile.src || "initial_element" : null;

      const input = findInputElement();
      if (!input) {
        console.error(`[${sceneId}] Input box missing. Retrying in ${RETRY_DELAY_MS / 1000}s...`);
        await delay(RETRY_DELAY_MS);
        continue;
      }

      injectTextReal(input, promptText);
      await delay(1000);
      triggerSubmit(input);
      console.log(`[${sceneId}] Submitted prompt. Waiting for tile...`);

      await delay(GENERATE_START_DELAY_MS);

      const finalTile = await waitForStableNewTile(initialKey);
      if (!finalTile) {
        console.warn(`[${sceneId}] No stable tile found on attempt ${attempt}.`);
        await delay(RETRY_DELAY_MS);
        continue;
      }

      const DOWNLOAD_RETRIES = 3;
      let downloaded = false;
      let currentTile = finalTile;

      for (let d = 1; d <= DOWNLOAD_RETRIES; d++) {
        downloaded = await downloadMasterImage(currentTile, `${sceneId}.png`);
        if (downloaded) break;
        await delay(2000);
        const refreshed = getTopLeftTile();
        if (refreshed) currentTile = refreshed;
      }

      if (downloaded) {
        console.log(`%c[${sceneId}] Successfully generated and saved!`, "color:#00ff88;font-weight:bold;");
        return true;
      }

      console.warn(`[${sceneId}] Download failed on attempt ${attempt}.`);
      await delay(RETRY_DELAY_MS);
    }

    // Exceeded max attempts: Skip scene
    console.error(`%c[${sceneId}] FAILED after ${MAX_SCENE_ATTEMPTS} attempts. Skipping to next scene...`, "color:#ff4444;font-weight:bold;");
    return false;
  }

  // ---------------------------------------------------------------------
  // MAIN PIPELINE
  // ---------------------------------------------------------------------
  const results = loadResults();

  for (const { id, prompt } of SCENES) {
    if (results[id]?.status === "done") {
      console.log(`[${id}] Already done — skipping.`);
      continue;
    }

    if (results[id]?.status === "failed") {
      console.warn(`[${id}] Previously marked failed — skipping.`);
      continue;
    }

    const success = await processScene(id, prompt);
    if (success) {
      results[id] = { status: "done", completedAt: new Date().toISOString() };
    } else {
      results[id] = { status: "failed", failedAt: new Date().toISOString() };
    }
    saveResults(results);

    await delay(2500);
  }

  console.log("%c🏁 Process finished for all available scenes.", "color:#00ff88;font-weight:bold;font-size:16px;");
  downloadResultsSummary(results);
})();
